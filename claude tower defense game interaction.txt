  142         for (const e of enemies) {
  143             const dx = e.x - t.cx, dy = e.y - t.cy;
  144             if (dx * dx + dy * dy <= TOWER_RANGE * TOWER_RANGE && e.x > bestX) {
  145                 target = e;
  146                 bestX = e.x;
  147             }
  148         }
  149         t.target = target;
  150         if (target && t.cooldown <= 0) {
  151             t.cooldown = TOWER_RATE;
  152             const dx = target.x - t.cx, dy = target.y - t.cy;
  153             const len = Math.hypot(dx, dy);
  154             bullets.push({ x: t.cx, y: t.cy, vx: dx / len * 7, vy: dy / len * 7, target });
  155         }
  156     }
  157
  158     for (let i = bullets.length - 1; i >= 0; i--) {
  159         const b = bullets[i];
  160         if (!enemies.includes(b.target)) { bullets.splice(i, 1); continue; }
  161         const dx = b.target.x - b.x, dy = b.target.y - b.y;
  162         const len = Math.hypot(dx, dy);
  163         b.vx = dx / len * 7;
  164         b.vy = dy / len * 7;
  165         b.x += b.vx;
  166         b.y += b.vy;
  167         if (len < 8) {
  168             b.target.hp--;
  169             for (let s = 0; s < 6; s++) {
  170                 sparks.push({
  171                     x: b.target.x, y: b.target.y,
  172                     vx: (Math.random() - 0.5) * 5,
  173                     vy: (Math.random() - 0.5) * 5 - 1,
  174                     life: 16,
  175                     color: Math.random() < 0.5 ? '#ff5533' : '#ffcc00'
  176                 });
  177             }
  178             if (b.target.hp <= 0) {
  179                 gold += b.target.reward;
  180                 enemies.splice(enemies.indexOf(b.target), 1);
  181                 updateHUD();
  182             }
  183             bullets.splice(i, 1);
  184         }
  185     }
  186
  187     for (let i = sparks.length - 1; i >= 0; i--) {
  188         const s = sparks[i];
  189         s.x += s.vx; s.y += s.vy; s.vy += 0.2;
  190         s.life--;
  191         if (s.life <= 0) sparks.splice(i, 1);
  192     }
  193

  195         waveActive = false;
  196         if (wave >= 3) { gameOver = true; won = true; }
  197         updateHUD();
  198     }
  199 }
